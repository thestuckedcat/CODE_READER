cmake_minimum_required(VERSION 3.20)
project(atlas_semantic LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 18)
  message(FATAL_ERROR "This extractor is validated with LLVM/Clang 18")
endif()
find_path(CLANG_INCLUDE_DIR clang/Analysis/CFG.h HINTS ${LLVM_INCLUDE_DIRS} REQUIRED)
find_library(CLANG_CPP_LIBRARY NAMES clang-cpp HINTS ${LLVM_LIBRARY_DIRS} REQUIRED)
add_executable(atlas-semantic extractor.cpp)
target_compile_features(atlas-semantic PRIVATE cxx_std_17)
target_include_directories(atlas-semantic SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIR})
target_link_libraries(atlas-semantic PRIVATE ${CLANG_CPP_LIBRARY} LLVM)
install(TARGETS atlas-semantic RUNTIME DESTINATION bin)
